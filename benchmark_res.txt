### Komponent React: Formularz Danych do NDA (LexPortal UI)

Poniższy komponent służy do wprowadzania danych programisty z rygorystyczną walidacją NIP/PESEL.

```tsx
import React, { useState } from 'react';

interface NDAData {
  companyName: string;
  nip: string;
  pesel: string;
  isCompany: boolean;
}

const NDAGenerator: React.FC = () => {
  const [data, setData] = useState<NDAData>({
    companyName: '',
    nip: '',
    pesel: '',
    isCompany: true,
  });

  const [errors, setErrors] = useState<{ nip?: string; pesel?: string }>({});

  // Walidacja Regex zgodna ze standardem LexPortal
  const validate = (field: string, value: string) => {
    if (field === 'nip') {
      const nipRegex = /^\d{10}$/;
      setErrors(prev => ({
        ...prev,
        nip: nipRegex.test(value) ? undefined : 'NIP musi składać się z 10 cyfr.'
      }));
    }
    if (field === 'pesel') {
      const peselRegex = /^\d{11}$/;
      setErrors(prev => ({
        ...prev,
        pesel: peselRegex.test(value) ? undefined : 'PESEL musi składać się z 11 cyfr.'
      }));
    }
  };

  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const { name, value } = e.target;
    setData({ ...data, [name]: value });
    validate(name, value);
  };

  return (
    <div className="p-8 max-w-2xl mx-auto bg-slate-50 rounded-xl shadow-lg border border-slate-200">
      <h2 className="text-xl font-bold text-indigo-900 mb-6 italic underline">Generator Danych Zleceniobiorcy (NDA)</h2>
      
      <div className="space-y-4">
        <div>
          <label className="block text-sm font-bold text-slate-700">Nazwa Wykonawcy / Programisty</label>
          <input
            type="text"
            name="companyName"
            className="w-full p-2 border border-slate-300 rounded focus:ring-2 focus:ring-indigo-500 outline-none"
            onChange={handleChange}
          />
        </div>

        <div className="grid grid-cols-2 gap-4">
          <div>
            <label className="block text-sm font-bold text-slate-700">NIP (Zleceniobiorca B2B)</label>
            <input
              type="text"
              name="nip"
              className={`w-full p-2 border rounded outline-none transition-all ${errors.nip ? 'border-red-500 bg-red-50' : 'border-slate-300 focus:ring-2 focus:ring-indigo-500'}`}
              onChange={handleChange}
              placeholder="10 cyfr"
            />
            {errors.nip && <p className="text-red-500 text-[10px] mt-1 font-semibold italic">{errors.nip}</p>}
          </div>
          <div>
            <label className="block text-sm font-bold text-slate-700">PESEL (Osoba fizyczna)</label>
            <input
              type="text"
              name="pesel"
              className={`w-full p-2 border rounded outline-none transition-all ${errors.pesel ? 'border-red-500 bg-red-50' : 'border-slate-300 focus:ring-2 focus:ring-indigo-500'}`}
              onChange={handleChange}
              placeholder="11 cyfr"
            />
            {errors.pesel && <p className="text-red-500 text-[10px] mt-1 font-semibold italic">{errors.pesel}</p>}
          </div>
        </div>

        {/* Penalty Cap Info Section */}
        <div className="mt-6 p-4 bg-amber-100 border-l-4 border-amber-500 rounded text-xs text-amber-900">
          <p className="font-bold uppercase mb-1 underline text-[10px]">Pouczenie o limicie odpowiedzialności (Penalty Cap):</p>
          Zgodnie z polityką LexPortal, niniejszy wzór zawiera limit odpowiedzialności Wykonawcy. Standardowa kwota 
          została ustawiona na 50.000,00 PLN, aby zminimalizować ryzyko finansowe Zleceniobiorcy przy zachowaniu interesu Zleceniodawcy.
        </div>

        {/* GDPR Section */}
        <div className="mt-4 p-4 bg-blue-100 border-l-4 border-blue-500 rounded text-[10px] text-blue-900 leading-snug italic">
          <p className="font-bold uppercase mb-1">Nota RODO:</p>
          Administratorem Twoich danych osobowych jest Zleceniodawca (Zamawiający). Masz prawo do wglądu, 
          sprostowania oraz żądania usunięcia danych po zakończeniu okresu przedawnienia roszczeń z umowy NDA.
        </div>

        <button className="w-full bg-indigo-600 text-white font-bold py-3 rounded-lg hover:bg-indigo-700 shadow-md transform active:scale-95 transition-all uppercase tracking-widest text-sm">
          Przygotuj Szkic NDA
        </button>
      </div>
    </div>
  );
};

export default NDAGenerator;
```
